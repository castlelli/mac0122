/* ---- 8. Estruturas de controle ---------------------------------------- */
int controle(int n)
{
	int i, total = 0;	/* indentado com TAB */

    if (n < 0) {
        return -1;
    } else if (n == 0) {
        return 0;
    } else {
        total = 1;
    }

    for (i = 0; i < n; i++) {
        if (i % 2 == 0) continue;
        if (i > 50) break;
        total += i;
    }
    while (total > 1000) total /= 2;
    do { total++; } while (total < 10);

    switch (n) {
        case 1:
            total += 1;
            break;
        case 2: case 3:
            total += 2;
            /* continua */
        default:
            total += 3;
    }

    {
        auto int a = 1;
        register int r = 2;
        static int s = 3;
        total += a + r + s;
    }

    #ifdef MAX
    total += MAX;
    #endif

    goto fim;
fim:
    return total;
}
